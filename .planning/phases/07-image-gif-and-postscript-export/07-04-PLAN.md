class XvueExport {
public:

    // QMessageBox::critical only if interactive. The "interactive" boolean

// xvue/qt/src/xvue_qt_export.cpp
// Phase 7 Plan 04 (EXPORT-02, EXPORT-05): bodies.
#include "xvue_qt_export.h"
#include "xvue_qt_app.h"
#include "xvue_qt_canvas.h"
#include "xvue_qt_state.h"
#include "xvue_qt_window.h"
#include "xvue_qt_console_dock.h"
#include "xvue_qt_i18n.h"
#include "xvue_qt_macros.h"

#include <QFileDialog>
#include <QImage>
#include <QImageWriter>
#include <QMessageBox>
#include <QPageSize>
#include <QPainter>
#include <QPdfWriter>
#include <QPixmap>
#include <QSettings>
#include <QSizeF>
#include <QStatusBar>

namespace {
// Helper: get the backing pixmap from XvueApp's current canvas.
const QPixmap* currentBacking() {
    auto& slot = XvueApp::window_slot();
    if (!slot) return nullptr;
    auto* canvas = slot->canvas();
    if (!canvas) return nullptr;
    return canvas->backingPtr();   // Plan 04 adds this accessor on XvueCanvas
                                    // if not already present.
}

XvueState* currentState() {
    auto& slot = XvueApp::window_slot();
    if (!slot) return nullptr;
    return slot->state();
}

void logFailure(const QString& msg, bool interactive) {
    if (auto& slot = XvueApp::window_slot(); slot) {
        if (auto* dock = slot->consoleDock()) dock->log(msg);
        if (auto* sb = slot->statusBar())   sb->showMessage(msg, 5000);
    }
    if (interactive) {
        QMessageBox::critical(nullptr,
            xvueT(MsgId::ErrorMsgBoxTitle),
            msg);
    }
}

QString rememberDir(const QString& chosen) {
    QSettings s("MEFISTO", "xvue");
    if (!chosen.isEmpty()) {
        s.setValue("export/last_dir", QFileInfo(chosen).absolutePath());
    }
    return s.value("export/last_dir").toString();
}
}  // namespace

// ---- savePngTo ----
bool XvueExport::savePngTo(const QString& path, bool interactive) {
    XVUE_QT_ASSERT_MAIN_THREAD();
    XvueApp::ensure();
    const QPixmap* bg = currentBacking();
    if (!bg) {
        logFailure(xvueT(MsgId::ExportNoCanvas), interactive);
        return false;
    }
    QImage img = bg->toImage();
    if (!img.save(path, "PNG")) {
        logFailure(xvueT(MsgId::ExportPngFailed), interactive);
        return false;
    }
    return true;
}

// ---- saveJpegTo ----
bool XvueExport::saveJpegTo(const QString& path, bool interactive) {
    XVUE_QT_ASSERT_MAIN_THREAD();
    XvueApp::ensure();
    const QPixmap* bg = currentBacking();
    if (!bg) {
        logFailure(xvueT(MsgId::ExportNoCanvas), interactive);
        return false;
    }
    QImageWriter w(path, "jpeg");
    int q = QSettings("MEFISTO","xvue").value("export/jpeg_quality", 90).toInt();
    w.setQuality(q);
    if (!w.write(bg->toImage())) {
        logFailure(xvueT(MsgId::ExportJpegFailed), interactive);
        return false;
    }
    return true;
}

// ---- savePdfTo (Pitfall 7: logical dims for setPageSize, backing for drawPixmap) ----
bool XvueExport::savePdfTo(const QString& path, bool interactive) {
    XVUE_QT_ASSERT_MAIN_THREAD();
    XvueApp::ensure();
    const QPixmap* bg = currentBacking();
    XvueState*     st = currentState();
    if (!bg || !st) {
        logFailure(xvueT(MsgId::ExportNoCanvas), interactive);
        return false;
    }
    QPdfWriter w(path);
    w.setResolution(72);
    // CRITICAL: logical pixels (Pitfall 7).
    w.setPageSize(QPageSize(QSizeF(st->xpixels(), st->ypixels()),
                            QPageSize::Point));
    QPainter p(&w);
    if (!p.isActive()) {
        logFailure(xvueT(MsgId::ExportPdfFailed), interactive);
        return false;
    }
    p.drawPixmap(0, 0, *bg);
    p.end();
    return true;
}

// ---- onMenu* slots: QFileDialog -> saveXxxTo ----
void XvueExport::onMenuExportPng() {
    XVUE_QT_ASSERT_MAIN_THREAD();
    QSettings s("MEFISTO","xvue");
    QString lastDir = s.value("export/last_dir").toString();
    QString path = QFileDialog::getSaveFileName(
        nullptr,
        xvueT(MsgId::FileExportPngDialogTitle),
        lastDir + "/canvas.png",
        xvueT(MsgId::FileExportPngFilter));
    if (path.isEmpty()) return;
    rememberDir(path);
    savePngTo(path, /*interactive=*/true);
}

void XvueExport::onMenuExportJpeg() {
    XVUE_QT_ASSERT_MAIN_THREAD();
    QSettings s("MEFISTO","xvue");
    QString lastDir = s.value("export/last_dir").toString();
    QString path = QFileDialog::getSaveFileName(
        nullptr,
        xvueT(MsgId::FileExportJpegDialogTitle),
        lastDir + "/canvas.jpg",
        xvueT(MsgId::FileExportJpegFilter));
    if (path.isEmpty()) return;
    rememberDir(path);
    saveJpegTo(path, /*interactive=*/true);
}

void XvueExport::onMenuExportPdf() {
    XVUE_QT_ASSERT_MAIN_THREAD();
    QSettings s("MEFISTO","xvue");
    QString lastDir = s.value("export/last_dir").toString();
    QString path = QFileDialog::getSaveFileName(
        nullptr,
        xvueT(MsgId::FileExportPdfDialogTitle),
        lastDir + "/canvas.pdf",
        xvueT(MsgId::FileExportPdfFilter));
    if (path.isEmpty()) return;
    rememberDir(path);
    savePdfTo(path, /*interactive=*/true);
}
