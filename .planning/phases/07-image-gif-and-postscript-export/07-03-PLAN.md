// ===== Per-primitive emit helpers — VERBATIM xvuelc.c format strings ======
// Pitfall 1: format strings ARE the contract. No QString::number, no <<.
// All Y coordinates flipped via pyFlip() (xvue/README_COORDS.md).

// ---- line: xvuelc.c:1942-1985 (xvtrait PS branch) ----
void PsEmitter::line(int x1, int y1, int x2, int y2) {
    XVUE_QT_ASSERT_MAIN_THREAD();
    if (!active()) return;
    if (lasopsc_ < 3) {
        if (nbrcon_ == 0) {
            nbrcon_ = 1;
            xinic_ = x1; yinic_ = y1; xcouc_ = x2; ycouc_ = y2;
            if (counb_ != -1.0f) {
                std::snprintf(&concat_[0], sizeof concat_,
                    "%6i %6i %6i %6i %3i %4.2f %4.2f %4.2f %4.2f S\n",
                    x1, pyFlip(y1), x2, pyFlip(y2),
                    nbrcon_, courgb_[0], courgb_[1], courgb_[2], counb_);
            } else {
                std::snprintf(&concat_[0], sizeof concat_,
                    "%6i %6i %6i %6i %3i %4.2f %4.2f %4.2f 0.00 S\n",
                    x1, pyFlip(y1), x2, pyFlip(y2),
                    nbrcon_, courgb_[0], courgb_[1], courgb_[2]);
            }
        } else if (x1 == xcouc_ && y1 == ycouc_) {
            ++nbrcon_;
            if (x2 == xinic_ && y2 == yinic_) {
                iPo_ = 1;
                if (counb_ != -1.0f) {
                    std::snprintf(&concat_[std::strlen(concat_) - 26], 64,
                        "%3i %4.2f %4.2f %4.2f %4.2f P\n",
                        nbrcon_, courgb_[0], courgb_[1], courgb_[2], counb_);
                } else {
                    std::snprintf(&concat_[std::strlen(concat_) - 26], 64,
                        "%3i %4.2f %4.2f %4.2f 0.00 P\n",
                        nbrcon_, courgb_[0], courgb_[1], courgb_[2]);
                }
                std::fprintf(fpo_, "%s", concat_);
                nbrcon_ = 0;
                concat_[0] = '\0';
            } else if (nbrcon_ % 16 == 0) {
                // continuation arm — verbatim xvuelc.c:1985
                std::snprintf(&concat_[std::strlen(concat_) - 26], 4, "\n ");
                xcouc_ = x2; ycouc_ = y2;
            } else {
                xcouc_ = x2; ycouc_ = y2;
            }
        } else {
            // Different starting point — flush pending and restart.
            std::fprintf(fpo_, "%s", concat_);
            nbrcon_ = 1;
            xinic_ = x1; yinic_ = y1; xcouc_ = x2; ycouc_ = y2;
            if (counb_ != -1.0f) {
                std::snprintf(&concat_[0], sizeof concat_,
                    "%6i %6i %6i %6i %3i %4.2f %4.2f %4.2f %4.2f S\n",
                    x1, pyFlip(y1), x2, pyFlip(y2),
                    nbrcon_, courgb_[0], courgb_[1], courgb_[2], counb_);
            } else {
                std::snprintf(&concat_[0], sizeof concat_,
                    "%6i %6i %6i %6i %3i %4.2f %4.2f %4.2f 0.00 S\n",
                    x1, pyFlip(y1), x2, pyFlip(y2),
                    nbrcon_, courgb_[0], courgb_[1], courgb_[2]);
            }
        }
    } else {
        // chaine_[lasopsc_-4] menu path — index guarded (Plan 02 note).
        int idx = lasopsc_ - 4;
        if (idx >= 0 && idx < MXRECT_ && chaine_[idx] != nullptr) {
            char* dst = chaine_[idx] + std::strlen(chaine_[idx]);
            std::snprintf(dst, longchaine_[idx] - std::strlen(chaine_[idx]),
                "%6i %6i %6i %6i %3i %4.2f %4.2f %4.2f %4.2f S\n",
                x1, pyFlip(y1), x2, pyFlip(y2),
                1, courgb_[0], courgb_[1], courgb_[2],
                (counb_ != -1.0f) ? counb_ : 0.0f);
        }
    }
}

// ---- face: xvuelc.c:2050-2150 (xvface) ----
// Emits moveto + lineto sequence + "F\n" (close-and-fill).
void PsEmitter::face(const int* xy, int n) {
    XVUE_QT_ASSERT_MAIN_THREAD();
    if (!active()) return;
    if (n < 1) return;
    if (lasopsc_ < 3) {
        // flush any pending concat
        if (nbrcon_ > 0) {
            std::fprintf(fpo_, "%s", concat_);
            nbrcon_ = 0;
            concat_[0] = '\0';
        }
        std::fprintf(fpo_, "%6i %6i moveto\n", xy[0], pyFlip(xy[1]));
        for (int i = 1; i < n; ++i) {
            std::fprintf(fpo_, "%6i %6i lineto\n", xy[2*i], pyFlip(xy[2*i+1]));
        }
        std::fprintf(fpo_, "F\n");
    } else {
        // chaine_ path
        int idx = lasopsc_ - 4;
        if (idx >= 0 && idx < MXRECT_ && chaine_[idx] != nullptr) {
            char* dst = chaine_[idx] + std::strlen(chaine_[idx]);
            int rem  = longchaine_[idx] - (int)std::strlen(chaine_[idx]);
            int wrote = std::snprintf(dst, rem, "%6i %6i moveto\n", xy[0], pyFlip(xy[1]));
            if (wrote > 0) { dst += wrote; rem -= wrote; }
            for (int i = 1; i < n && rem > 32; ++i) {
                int w = std::snprintf(dst, rem, "%6i %6i lineto\n", xy[2*i], pyFlip(xy[2*i+1]));
                if (w > 0) { dst += w; rem -= w; }
            }
            std::snprintf(dst, rem, "F\n");
        }
    }
}

}

// ---- chargefonte: D-08 hardcoded mapping table — replaces xvuelc.c:1500-1568
// X11 font-name string-bash. The QFont::family() value comes from the Qt-side
// xvchargefonte_ caller (it was already loading a QFont — Plan 03 reads
// QFont::family() at the call site and passes the QString here).
void PsEmitter::chargefonte(const QString& family, int size_pt,
                             int /*ascent*/, int /*descent*/,
                             bool bold, bool italic) {
    XVUE_QT_ASSERT_MAIN_THREAD();
    if (!active()) return;

    // D-08: 4-entry mapping table.
    QString ps_family;
    bool use_italic_keyword = false;  // /Times uses "Italic"; others use "Oblique"
    if (family == "Courier") {
        ps_family = "/Courier";
    } else if (family == "Times" || family == "Times-Roman" || family == "Times New Roman") {
        ps_family = "/Times-Roman";
        use_italic_keyword = true;
    } else if (family == "Helvetica") {
        ps_family = "/Helvetica";
    } else if (family == "New Century Schoolbook" || family == "NewCenturySchlbk") {
        ps_family = "/NewCenturySchlbk";
        use_italic_keyword = true;
    } else {
        // Fallback: warn-once, route to /Helvetica per D-08.
        static bool warned = false;
        if (!warned) {
            warned = true;
            std::fprintf(stderr,
                "[PsEmitter] Font family '%s' has no PostScript mapping; "
                "falling back to /Helvetica (D-08 fallback)\n",
                qPrintable(family));
        }
        ps_family = "/Helvetica";
    }

    // Bold / Italic suffix (xvuelc.c:1530-1565 logic, faithfully ported).
    QString suffix;
    if (bold && italic) {
        suffix = use_italic_keyword ? "-BoldItalic" : "-BoldOblique";
    } else if (bold) {
        suffix = "-Bold";
    } else if (italic) {
        suffix = use_italic_keyword ? "-Italic" : "-Oblique";
    }

    QString full = ps_family + suffix;
    std::strncpy(fontcour_, qPrintable(full), sizeof(fontcour_) - 1);

    if (lasopsc_ < 3) {
        std::fprintf(fpo_, "%s findfont %i scalefont setfont\n",
            fontcour_, size_pt);
    } else {
        int idx = lasopsc_ - 4;
        if (idx >= 0 && idx < MXRECT_ && chaine_[idx] != nullptr) {
            char* dst = chaine_[idx] + std::strlen(chaine_[idx]);
            std::snprintf(dst, longchaine_[idx] - std::strlen(chaine_[idx]),
                "%s findfont %i scalefont setfont\n", fontcour_, size_pt);
        }
    }
}

// ---- texte: xvuelc.c:~3100-3160 ----
void PsEmitter::texte(const char* s, int slen, int x, int y) {
    XVUE_QT_ASSERT_MAIN_THREAD();
    if (!active() || s == nullptr || slen <= 0) return;
    // Escape ()\\ per PostScript string literal rules.
    char escaped[512];
    int j = 0;
    for (int i = 0; i < slen && j < (int)sizeof(escaped) - 2; ++i) {
        if (s[i] == '(' || s[i] == ')' || s[i] == '\\') {
            escaped[j++] = '\\';
        }
        escaped[j++] = s[i];
    }
    escaped[j] = '\0';
    if (lasopsc_ < 3) {
        std::fprintf(fpo_, "%6i %6i moveto\n", x, pyFlip(y));
        std::fprintf(fpo_, "(%s) show\n", escaped);
    } else {
        int idx = lasopsc_ - 4;
        if (idx >= 0 && idx < MXRECT_ && chaine_[idx] != nullptr) {
            char* dst = chaine_[idx] + std::strlen(chaine_[idx]);
            int rem = longchaine_[idx] - (int)std::strlen(chaine_[idx]);
            int w = std::snprintf(dst, rem, "%6i %6i moveto\n", x, pyFlip(y));
            if (w > 0) { dst += w; rem -= w; }
            std::snprintf(dst, rem, "(%s) show\n", escaped);
        }
    }
}
