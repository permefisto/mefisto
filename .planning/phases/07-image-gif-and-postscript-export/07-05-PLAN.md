```cpp
// ---- Plan 05 implementations ----
#include <QImage>
#include <QImageWriter>
#include <QList>
#include <QProcess>
#include <QStandardPaths>
#include <QTemporaryDir>
#include <QFileInfo>
#include <QDir>

namespace {
// Process-wide singletons. Main-thread-only.
bool g_capture_active = false;
QList<QImage> g_pending_frames;
int g_warned_frames_softcap = 0;   // sticky flag — warn-once per beginAnimation
#ifdef XVUE_QT_TESTING
int  g_ffmpeg_forced_exit = -1;    // -1 = use real ffmpeg
bool g_native_gif_forced  = false; // override usingNativeGifWriter()
#endif

QString tempDirRoot() {
    return QStandardPaths::writableLocation(QStandardPaths::TempLocation);
}
}  // namespace

bool XvueExport::isCaptureActive() {
    return g_capture_active;
}

int XvueExport::pendingFrameCount() {
    return static_cast<int>(g_pending_frames.size());
}

void XvueExport::beginAnimation() {
    XVUE_QT_ASSERT_MAIN_THREAD();
    XvueApp::ensure();
    g_capture_active = true;
    g_pending_frames.clear();
    g_warned_frames_softcap = 0;
    if (auto& slot = XvueApp::window_slot(); slot) {
        if (auto* sb = slot->statusBar()) {
            sb->showMessage(xvueT(MsgId::AnimationStarted), 3000);
        }
    }
}

void XvueExport::captureFrame() {
    XVUE_QT_ASSERT_MAIN_THREAD();
    if (!g_capture_active) return;
    const QPixmap* bg = currentBacking();
    if (!bg) return;

    // T-07-03: hard cap.
    if ((int)g_pending_frames.size() >= kAnimFrameHardCap) {
        if (auto& slot = XvueApp::window_slot(); slot) {
            if (auto* dock = slot->consoleDock())
                dock->log(xvueT(MsgId::AnimationFrameHardCapHit));
        }
        // Force end-of-animation — push remaining frames to disk.
        endAnimation();
        return;
    }

    g_pending_frames.append(bg->toImage());

    // T-07-03: soft warn at 100 (warn-once).
    if (!g_warned_frames_softcap && (int)g_pending_frames.size() >= kAnimFrameSoftCap) {
        g_warned_frames_softcap = 1;
        if (auto& slot = XvueApp::window_slot(); slot) {
            if (auto* dock = slot->consoleDock())
                dock->log(xvueT(MsgId::AnimationFrameSoftCapWarn));
        }
    }
}

void XvueExport::endAnimation() {
    XVUE_QT_ASSERT_MAIN_THREAD();
    if (!g_capture_active) return;
    g_capture_active = false;
    if (g_pending_frames.isEmpty()) return;

    // Default output location: cwd/animation.gif (D-03 legacy compat).
    QString defaultPath = QDir::current().filePath("animation.gif");
    saveGifTo(defaultPath, /*interactive=*/false);
}

bool XvueExport::usingNativeGifWriter() {
#ifdef XVUE_QT_TESTING
    if (g_native_gif_forced) return true;
#endif
    auto formats = QImageWriter::supportedImageFormats();
    for (const auto& f : formats) {
        if (f.toLower() == "gif") return true;
    }
    return false;
}

bool XvueExport::saveGifTo(const QString& outputPath, bool interactive) {
    XVUE_QT_ASSERT_MAIN_THREAD();
    if (g_pending_frames.isEmpty()) {
        logFailure(xvueT(MsgId::AnimationNoFrames), interactive);
        return false;
    }

    // Status-bar feedback (D-11).
    if (auto& slot = XvueApp::window_slot(); slot) {
        if (auto* sb = slot->statusBar()) {
            sb->showMessage(xvueT(MsgId::AnimationEncoding), 0);
        }
    }

    bool ok = false;

    if (usingNativeGifWriter()) {
        // ----- D-10 native QImageWriter multi-frame path (defensive) -----
        // Per PROBE.md this path is dead today on Debian/Qt 6.10.2 but kept
        // for portability if a future Debian add-on enables GIF write.
        QImageWriter w(outputPath, "gif");
        int delay = QSettings("MEFISTO","xvue").value("export/gif_delay", 10).toInt();
        w.setText("Delay", QString::number(delay));
        // Multi-frame write loop:
        bool any_failed = false;
        for (const auto& img : g_pending_frames) {
            if (!w.write(img)) { any_failed = true; break; }
        }
        ok = !any_failed;
    } else {
        // ----- D-11 ffmpeg fallback path (the realized path on this host) -----
        QTemporaryDir tempDir(QDir(tempDirRoot()).filePath("xvue-gif-XXXXXX"));
        if (!tempDir.isValid()) {
            logFailure(xvueT(MsgId::AnimationTempDirFailed), interactive);
            g_pending_frames.clear();
            return false;
        }

        // Write PNG sequence frame_NNNN.png (NNNN starts at 0001).
        bool any_failed = false;
        int idx = 0;
        for (const auto& img : g_pending_frames) {
            ++idx;
            QString fname = tempDir.filePath(
                QString("frame_%1.png").arg(idx, 4, 10, QChar('0')));
            if (!img.save(fname, "PNG")) { any_failed = true; break; }
        }
        if (any_failed) {
            tempDir.setAutoRemove(false);  // T-07-05: keep on failure
            if (auto& slot = XvueApp::window_slot(); slot) {
                if (auto* dock = slot->consoleDock())
                    dock->log(QString("Frame write failed; tempdir kept at %1")
                              .arg(tempDir.path()));
            }
            g_pending_frames.clear();
            return false;
        }

        // ffmpeg dispatch (T-07-04: argv is constants + temp paths only).
        int delay = QSettings("MEFISTO","xvue").value("export/gif_delay", 10).toInt();
        QStringList args{
            "-y",
            "-framerate", QString::number(delay),
            "-i", tempDir.filePath("frame_%04d.png"),
            outputPath
        };
#ifdef XVUE_QT_TESTING
        int rc = (g_ffmpeg_forced_exit >= 0)
                 ? g_ffmpeg_forced_exit
                 : QProcess::execute("ffmpeg", args);
#else
        int rc = QProcess::execute("ffmpeg", args);
#endif
        if (rc != 0) {
            tempDir.setAutoRemove(false);  // T-07-05: keep on failure
            QString msg = xvueT(MsgId::AnimationFfmpegFailed)
                          + QString(" (rc=%1, tempdir=%2)").arg(rc).arg(tempDir.path());
            logFailure(msg, interactive);
            g_pending_frames.clear();
            return false;
        }
        ok = true;
        // tempDir auto-removed by RAII destructor on success (T-07-05).
    }

    g_pending_frames.clear();
    g_warned_frames_softcap = 0;

    if (auto& slot = XvueApp::window_slot(); slot) {
        if (auto* sb = slot->statusBar()) {
            sb->showMessage(ok ? xvueT(MsgId::AnimationDone)
                               : xvueT(MsgId::AnimationFailed), 5000);
        }
    }
    if (!ok) {
        logFailure(xvueT(MsgId::AnimationFailed), interactive);
    }
    return ok;
}

void XvueExport::onMenuExportGif() {
    XVUE_QT_ASSERT_MAIN_THREAD();
    if (!isCaptureActive() && g_pending_frames.isEmpty()) {
        logFailure(xvueT(MsgId::AnimationNoFrames), /*interactive=*/true);
        return;
    }
    QSettings s("MEFISTO","xvue");
    QString lastDir = s.value("export/last_dir").toString();
    QString path = QFileDialog::getSaveFileName(
        nullptr,
        xvueT(MsgId::FileExportGifDialogTitle),
        lastDir + "/animation.gif",
        xvueT(MsgId::FileExportGifFilter));
    if (path.isEmpty()) return;
    rememberDir(path);
    if (isCaptureActive()) g_capture_active = false;  // close capture
    saveGifTo(path, /*interactive=*/true);
}

void XvueExport::onMenuToggleCapture() {
    XVUE_QT_ASSERT_MAIN_THREAD();
    if (isCaptureActive()) endAnimation();
    else                   beginAnimation();
}

void XvueExport::checkEnvAutoStart() {
    if (qgetenv("XVUE_ANIM").trimmed() == "1") {
        beginAnimation();
    }
}

#ifdef XVUE_QT_TESTING
void XvueExport::resetForTesting() {
    g_capture_active = false;
    g_pending_frames.clear();
    g_warned_frames_softcap = 0;
    g_ffmpeg_forced_exit = -1;
    g_native_gif_forced = false;
}
void XvueExport::setNativeGifWriterForTesting(bool v) { g_native_gif_forced = v; }
void XvueExport::setFfmpegOverrideForTesting(int rc)  { g_ffmpeg_forced_exit = rc; }
#endif
```
