       if [ -z "$MEFISTO" ]; then
           SCRIPT_DIR="$(cd "$(dirname "$0")" && pwd)"
           REPO_ROOT="$(dirname "$SCRIPT_DIR")"
           [ -d "$REPO_ROOT/xvue/qt" ] && MEFISTO="$REPO_ROOT" || { \

           xvue/*.f flui/ ther/ util/ prpr/ 2>/dev/null \

           || true)
       if [ -n "$CONVERT_HITS" ]; then

       fi
