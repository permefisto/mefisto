```cpp
void testMenuOrder() {
    auto* win = XvueApp::window_slot().get();
    QVERIFY(win);
    auto* mb = win->menuBar();
    QVERIFY(mb);
    QList<QMenu*> topMenus;
    for (QAction* a : mb->actions()) {
        if (a->menu()) topMenus.append(a->menu());
    }
    QVERIFY2(topMenus.size() >= 4,
             qPrintable(QStringLiteral("Expected at least 4 top-level menus, got %1")
                        .arg(topMenus.size())));
    QCOMPARE(topMenus[0]->objectName(), QStringLiteral("File"));
    QCOMPARE(topMenus[1]->objectName(), QStringLiteral("Thermal"));
    QCOMPARE(topMenus[2]->objectName(), QStringLiteral("View"));
    QCOMPARE(topMenus[3]->objectName(), QStringLiteral("Help"));
}
```

```cpp
void testBilingualLabelsEnglish() {
    const QString mefisto = qEnvironmentVariable("MEFISTO");
    if (mefisto.isEmpty()) QSKIP("MEFISTO env not set");
    if (!QFile::exists(mefisto + QStringLiteral("/td/m/anglais")))
        QSKIP("anglais flag not set (FR install)");
    if (!QFile::exists(mefisto + QStringLiteral("/td/ma/debuther")))
        QSKIP("td/ma/debuther not found");
    XvueMenuFileParser::clearCacheForTesting();
    const MenuFile& mf = XvueMenuFileParser::loadFor(QStringLiteral("debuther"));
    QVERIFY(mf.ok());
    // Verify EN labels for canonical thermal codes per td/ma/debuther:
    //   3 → "STEADY HEAT TRANSFER solver"
    //   8 → "DRAWING of TEMPERATURES and FLUX"
    //   99 → "SAVE DATA and QUIT"
    const QString label3 = mf.label(3);
    QVERIFY2(label3.contains(QStringLiteral("STEADY"), Qt::CaseInsensitive)
             || label3.contains(QStringLiteral("HEAT"), Qt::CaseInsensitive),
             qPrintable(QStringLiteral("Expected EN steady-heat label for 3;, got: ") + label3));
    const QString label8 = mf.label(8);
    QVERIFY2(label8.contains(QStringLiteral("TEMPERATURE"), Qt::CaseInsensitive)
             || label8.contains(QStringLiteral("FLUX"), Qt::CaseInsensitive),
             qPrintable(QStringLiteral("Expected EN temperature label for 8;, got: ") + label8));
    const QString label99 = mf.label(99);
    QVERIFY2(label99.contains(QStringLiteral("SAVE"), Qt::CaseInsensitive)
             || label99.contains(QStringLiteral("QUIT"), Qt::CaseInsensitive),
             qPrintable(QStringLiteral("Expected EN save/quit label for 99;, got: ") + label99));
    XvueMenuFileParser::clearCacheForTesting();
}
```

```cpp
class TestXvueQtTherMenu : public QObject {
    Q_OBJECT

private:
    // Recursive helper — find a QAction whose "lexicon" dynamic property
    // matches `lex`. Walks submenus. Same as 6.1/6.2/6.3 pattern.
    static QAction* actionByLexicon(QMenu* menu, const QString& lex) {
        if (!menu) return nullptr;
        for (QAction* a : menu->actions()) {
            if (a->property("lexicon").toString() == lex) return a;
            if (a->menu()) {
                if (auto* child = actionByLexicon(a->menu(), lex)) return child;
            }
        }
        return nullptr;
    }

    // Drain chars via xvsouris_ and assert the sequence.
    static void expectChars(const char* expected, int n) {
        for (int i = 0; i < n; ++i) {
            int nt = 0, nbc = 0, x = 0, y = 0;
            xvsouris_(&nt, &nbc, &x, &y);
            QCOMPARE(nt, 2);
            QCOMPARE(nbc, static_cast<int>(static_cast<unsigned char>(expected[i])));
        }
    }

private slots:

        qunsetenv("MEFISTO_XVSOURIS_AUTOEXIT");
        xvinitgraphique_();
        char name[] = "ther";
        int  nlen   = 4;
        xvue_module_init_(name, &nlen);  // registerTherActions_stub_ runs
    }

    void cleanupTestCase() {
        auto* win = XvueApp::window_slot().get();
        if (win && win->menuBridge()) {
            while (win->menuBridge()->popChar().has_value()) {}
        }
        xvfermer_();
    }

    void testThermalSolveSteady() {
        auto* win = XvueApp::window_slot().get();
        QVERIFY(win && win->menuBridge());
        auto* thermalMenu = win->menuBar()->findChild<QMenu*>(QStringLiteral("Thermal"));
        QVERIFY(thermalMenu);
        auto* act = actionByLexicon(thermalMenu, QStringLiteral("3;"));
        QVERIFY(act);
        act->trigger();
        const char exp[3] = { '3', ';', 13 };
        expectChars(exp, 3);
    }

        expectChars(exp, 4);

    // Per 6.3 Plan 03 Auto-fix Rule 1: per-module Help-lexicon allowlist
    // drawn from LEXICON-AUDIT-ther.md row 98. NOT `{97;}` like flui!
    void testHelpNoQueue() {

        auto* help = win->menuBar()->findChild<QMenu*>(QStringLiteral("Help"));
        QVERIFY(help);
        const int before = win->menuBridge()->queueSize();
        const QSet<QString> allowlist{ QStringLiteral("98;") };
        for (QAction* a : help->actions()) {
            const QString lex = a->property("lexicon").toString();
            if (lex.isEmpty()) continue;  // shared 6.0 generic Help — no audit
            QVERIFY2(allowlist.contains(lex),
                     qPrintable(QStringLiteral("help action %1 has unintended "
                                               "lexicon %2 (allowlist={98;})")
                                .arg(a->text(), lex)));
        }
        QCOMPARE(win->menuBridge()->queueSize(), before);
    }

    void testCloseEventPushes99WhenBlocking() {
        auto* win = XvueApp::window_slot().get();
        QVERIFY(win && win->menuBridge());
        while (win->menuBridge()->popChar().has_value()) {}
        const int before = win->menuBridge()->queueSize();
        {

            QCloseEvent ev;
            win->closeEvent(&ev);
            QVERIFY(!ev.isAccepted());

        QCOMPARE(after - before, 4);  // '9','9',';',CR
```
