#include "xvue_qt_console_dock.h"

#include <QSocketNotifier>

#include <QTextCursor>

#include <unistd.h>
#include <fcntl.h>
#include <cstdio>

XvueConsoleDock::XvueConsoleDock(QWidget* parent)
    : QDockWidget(xvueT(MsgId::ConsoleTitle), parent)
{
    textEdit_ = new QPlainTextEdit(this);
    textEdit_->setReadOnly(true);
    textEdit_->setMaximumBlockCount(10000);   // UI-SPEC cap
    textEdit_->setFont(QFontDatabase::systemFont(QFontDatabase::FixedFont));
    setWidget(textEdit_);

    errorBatcher_ = new XvueErrorBatcher(this);

    // Allowed dock areas per UI-SPEC §Dock area policy (no top).
    setAllowedAreas(Qt::BottomDockWidgetArea | Qt::LeftDockWidgetArea | Qt::RightDockWidgetArea);
}

XvueConsoleDock::~XvueConsoleDock() {
    if (notifier_) {
        notifier_->setEnabled(false);
    }
    // fd is leaked on purpose — closing the dup'd STDOUT_FILENO after redirect
    // would break any subsequent stdout write; the pipe lives for process life.
}

void XvueConsoleDock::installStdoutRedirect() {
    // Idempotent guard — only redirect once per process.
    if (readFd_ >= 0) return;

    int fd[2];
    if (::pipe(fd) != 0) {
        std::fprintf(stderr, "xvue-qt: pipe() failed; console dock disabled\n");
        return;
    }
    // T-06.0-CONSOLE-01 mitigation: expand pipe buffer from default 64K to 1MB.
    // Long Fortran mesh reports could back-pressure the main thread otherwise.
    (void)::fcntl(fd[0], F_SETPIPE_SZ, 1 << 20);  // 1 MB

    // Dup the write end onto STDOUT_FILENO so Fortran WRITE(IMPRIM,*) flows
    // into the pipe.
    if (::dup2(fd[1], STDOUT_FILENO) < 0) {
        std::fprintf(stderr, "xvue-qt: dup2 stdout failed; console dock disabled\n");
        ::close(fd[0]); ::close(fd[1]);
        return;
    }
    ::close(fd[1]);

    // Pitfall 3.4: setvbuf AFTER dup2 so it applies to the new stdout.
    std::setvbuf(stdout, nullptr, _IOLBF, 0);

    readFd_   = fd[0];
    notifier_ = new QSocketNotifier(readFd_, QSocketNotifier::Read, this);
    connect(notifier_, &QSocketNotifier::activated,
            this,      &XvueConsoleDock::onPipeReadable);
}

void XvueConsoleDock::appendLine(const QString& line) {
    if (!textEdit_) return;
    textEdit_->appendPlainText(line);
    // Auto-scroll to bottom.
    QTextCursor c = textEdit_->textCursor();
    c.movePosition(QTextCursor::End);
    textEdit_->setTextCursor(c);
}

void XvueConsoleDock::onPipeReadable() {
    if (readFd_ < 0) return;
    char buf[4096];
    ssize_t n;
    // Read until EAGAIN or we've drained the available bytes.
    while ((n = ::read(readFd_, buf, sizeof buf)) > 0) {
        partialLine_.append(buf, static_cast<int>(n));
        int pos;
        while ((pos = partialLine_.indexOf('\n')) >= 0) {
            const QByteArray line = partialLine_.left(pos);
            partialLine_.remove(0, pos + 1);
            const QString s = QString::fromUtf8(line);
            appendLine(s);
            if (s.startsWith("*** ERREUR") || s.startsWith("*** ERROR")) {
                if (errorBatcher_) errorBatcher_->enqueue(s);
            }
        }
        if (n < static_cast<ssize_t>(sizeof buf)) break;  // drained
    }
}
```

#include "xvue_qt_error_batcher.h"
#include "xvue_qt_app.h"
#include "xvue_qt_i18n.h"

#include <QTimer>
#include <QMessageBox>
#include <QCoreApplication>

XvueErrorBatcher::XvueErrorBatcher(QObject* parent) : QObject(parent) {
    timer_ = new QTimer(this);
    timer_->setSingleShot(true);
    timer_->setInterval(kBatchWindowMs);
    connect(timer_, &QTimer::timeout, this, &XvueErrorBatcher::flushErrorBatch);
}

XvueErrorBatcher::~XvueErrorBatcher() = default;

void XvueErrorBatcher::enqueue(const QString& errorLine) {
    pendingErrorLines_ << errorLine;
    if (!timer_->isActive()) {
        timer_->start();
    }
}

void XvueErrorBatcher::flushErrorBatch() {
    if (pendingErrorLines_.isEmpty()) return;

    // T-06-03-01: defer QMessageBox when a Fortran blocking read is in progress.
    // Nested-dialog-inside-xvsouris risks eating the deferred-quit timer.
    if (XvueApp::blockingDepth() > 0) {
        // Re-arm for another kBatchWindowMs window.
        timer_->start();
        return;
    }

    const QString body = pendingErrorLines_.join("\n");
    pendingErrorLines_.clear();

    // Non-blocking message box on a QObject parent would be cleaner, but
    // QMessageBox::warning is modal by convention. Wrap in a deferred call
    // via QMetaObject::invokeMethod with QueuedConnection so we don't block
    // inside the timer's slot.
    const QString title = xvueT(MsgId::ErrorMsgBoxTitle);
    QMetaObject::invokeMethod(qApp, [title, body]{
        QMessageBox::warning(nullptr, title, body);
    }, Qt::QueuedConnection);
}
```

Note on `qApp` includes: `<QCoreApplication>` makes `qApp` available. If `QMessageBox::warning` can't have `nullptr` parent on some platforms, change to pass the console dock's parent window. The queued-invoke ensures the message box fires outside the timer slot's execution context.

    clear();   // remove all existing QActions

    const QStringList recent = XvuePrefs::recentProjects();
    if (recent.isEmpty()) {
        // Show a greyed-out placeholder so the user knows the submenu is live.
        QAction* empty = addAction(tr("(none)"));
        empty->setEnabled(false);
    } else {
        for (const QString& path : recent) {
            const QString label = QFileInfo(path).fileName();
            // Tooltip = full path; menu label = basename for readability.
            QAction* a = addAction(label);
            a->setToolTip(path);
            connect(a, &QAction::triggered, this, [this, path]{
                emit openProjectRequested(path);
            });
        }
    }

    addSeparator();
    QAction* clear = addAction(xvueT(MsgId::FileRecentClear));
    connect(clear, &QAction::triggered, this, &XvueRecentProjectsMenu::onClearRequested);
}

void XvueRecentProjectsMenu::onClearRequested() {
    // UI-SPEC §Destructive actions: confirm with OK+Cancel.
    QMessageBox box(this->parentWidget());
    box.setIcon(QMessageBox::Question);
    box.setWindowTitle(xvueT(MsgId::FileRecentClearConfirmTitle));
    box.setText(xvueT(MsgId::DestructiveConfirmBodyGeneric));
    box.addButton(xvueT(MsgId::ButtonOk), QMessageBox::AcceptRole);
    box.addButton(xvueT(MsgId::ButtonCancel), QMessageBox::RejectRole);
    if (box.exec() == 0 /* first button = Ok/Accept */ ||
        box.result() == QMessageBox::Accepted) {
        XvuePrefs::clearRecentProjects();
        refresh();
    }
    // Simpler: use box.exec() and test result() against AcceptRole; Qt returns
    // the index of the button clicked.
}
