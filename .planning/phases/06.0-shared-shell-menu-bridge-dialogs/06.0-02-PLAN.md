void XvueMenuBridge::queueLexicon(const QString& cmd) {
    // Phase 6 UX-02. Main-thread-only (SHELL-07) — the bridge is a QObject
    // parented under XvueWindow which lives on the main thread.
    Q_ASSERT(QThread::currentThread() == qApp->thread());

    // T-06.0-BRIDGE-01: cap queue at kMaxQueueSize. If a user clicks menu
    // items faster than Fortran drains, dropping is the recoverable failure
    // mode — unbounded growth would segfault. cmd + trailing CR counts as
    // cmd.length()+1 items.
    const int wouldBe = pendingChars_.size() + cmd.length() + 1;
    if (wouldBe > kMaxQueueSize) {
        // Silently drop. Phase 6 could surface a status-bar warning; UI-SPEC
        // does not require it, and a filling queue is a symptom of a stuck
        // Fortran loop, not a user error.
        return;
    }

    // Research §6 + saclav.f verification: each char goes as one QQueue<char>
    // element; trailing CR (=13) is the Fortran line-terminator that flushes
    // SACLAV's KLG(LHKLG) accumulator.
    for (QChar ch : cmd) {
        pendingChars_.enqueue(ch.toLatin1());
    }
    pendingChars_.enqueue(static_cast<char>(13));
}

std::optional<char> XvueMenuBridge::popChar() {
    Q_ASSERT(QThread::currentThread() == qApp->thread());
    if (pendingChars_.isEmpty()) return std::nullopt;
    return pendingChars_.dequeue();
}

void XvueMenuBridge::setContextMenuPopulator(ContextPopulator fn) {
    contextPopulator_ = std::move(fn);
}

void XvueMenuBridge::populateContextMenu(QMenu* m) {
    if (m && contextPopulator_) contextPopulator_(m);
}

QString XvuePrefs::moduleGroup_;

namespace {
// Centralized QSettings factory. Qt documents: constructing QSettings is

QSettings settings() {

QString scoped(const QString& key) {
    // Per-module key: "mail/window/geometry" etc.
    if (XvuePrefs_moduleGroup().isEmpty()) return key;
    return XvuePrefs_moduleGroup() + "/" + key;
}
QString XvuePrefs_moduleGroup() { return XvuePrefs::moduleGroup_; }
}

void XvuePrefs::initialize(const char* moduleName) {
    // Idempotent — safe to call from every pp*_qt main().
    QCoreApplication::setOrganizationName(QStringLiteral("LJLL"));
    QCoreApplication::setApplicationName(QStringLiteral("mefisto"));

bool XvuePrefs::consoleVisible(bool fallback) {
    try {
        return settings().value(scoped("window/console-visible"), fallback).toBool();
    } catch (...) { return fallback; }

    settings().setValue(scoped("window/console-visible"), v);
}

QStringList XvuePrefs::recentProjects() {
    try {
        QStringList list = settings().value(scoped("recent-projects"), QStringList()).toStringList();
        // D-06: prune non-existent silently at read time.
        QStringList pruned;
        for (const QString& p : list) {
            if (!p.isEmpty() && QFileInfo(p).exists()) pruned << p;
        }
        return pruned;
    } catch (...) { return QStringList(); }
}
void XvuePrefs::pushRecentProject(const QString& absPath) {
    if (absPath.isEmpty()) return;
    QStringList list = recentProjects();
    list.removeAll(absPath);   // dedupe
    list.prepend(absPath);
    while (list.size() > 10) list.removeLast();
    settings().setValue(scoped("recent-projects"), list);
}
void XvuePrefs::clearRecentProjects() {
    settings().setValue(scoped("recent-projects"), QStringList());
}

QString XvuePrefs::colorScheme() {
    try {
        const QString v = settings().value(QStringLiteral("ui/color-scheme"),
                                           QStringLiteral("system")).toString();
        // T-06.0-SETTINGS-01: clamp unknown → "system".
        if (v == QLatin1String("system") || v == QLatin1String("light") || v == QLatin1String("dark")) {
            return v;
        }
        return QStringLiteral("system");
    } catch (...) { return QStringLiteral("system"); }
}
void XvuePrefs::saveColorScheme(const QString& s) {
    // Accept only valid values — reject silently otherwise.
    if (s == QLatin1String("system") || s == QLatin1String("light") || s == QLatin1String("dark")) {
        settings().setValue(QStringLiteral("ui/color-scheme"), s);
    }
}
