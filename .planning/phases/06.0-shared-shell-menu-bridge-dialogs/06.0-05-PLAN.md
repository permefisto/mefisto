void XvueCanvas::paintEvent(QPaintEvent* event) {
    Q_UNUSED(event);
    QPainter p(this);

    // Phase 6.0 Plan 05 (UX-12): apply view transform BEFORE drawing backing.
    // DRAW-01 Phase 6 extension — paintEvent still does ONE drawPixmap, just
    // with a pre-applied transform.
    if (state_) {
        if (state_->backing_) {
            p.setTransform(state_->view_transform_);
            p.drawPixmap(0, 0, *state_->backing_);
        }
        // Reset transform for the empty-state text overlay so text is centered
        // in widget coords regardless of view_transform_.
        p.resetTransform();

        // Phase 6.0 Plan 05 + UI-SPEC Flag #3: empty-state hint.
        lastPaintDrewEmptyState_ = false;
        if (!state_->has_user_content_) {
            p.setPen(palette().color(QPalette::WindowText));
            // Body role (no explicit size change — use QApplication::font()).
            const QRect r = rect();
            QFontMetrics fm(p.font());
            const QString heading = xvueT(MsgId::EmptyStateHeading);
            const QString body    = xvueT(MsgId::EmptyStateBody);
            const int lineH = fm.height();
            const int headW = fm.horizontalAdvance(heading);
            const int bodyW = fm.horizontalAdvance(body);
            const int headX = (r.width() - headW) / 2;
            const int bodyX = (r.width() - bodyW) / 2;
            const int centerY = r.height() / 2;
            p.drawText(headX, centerY - lineH / 2, heading);
            p.drawText(bodyX, centerY + lineH,    body);
            lastPaintDrewEmptyState_ = true;
        }
    }
}

// (resizeEvent UNCHANGED)

void XvueCanvas::wheelEvent(QWheelEvent* event) {
    // UX-12 + Research §7. 1.15x per notch. angleDelta().y() = 120 per notch.
    const int notches = event->angleDelta().y() / 120;
    if (notches == 0 || !state_) {
        event->ignore();
        return;
    }
    const qreal factor = std::pow(1.15, notches);
    QTransform t = state_->view_transform_ * QTransform().scale(factor, factor);
    // Clamp m11 / m22 to [0.1, 10.0] to prevent extreme zoom.
    const qreal s = t.m11();
    if (s < 0.1 || s > 10.0) {
        event->accept();
        return;   // refuse zoom step that would exceed bounds
    }
    state_->view_transform_ = t;
    update();
    event->accept();
}

void XvueCanvas::mousePressEvent(QMouseEvent* event) {
    if (event->button() == Qt::MiddleButton && state_) {
        // Start middle-drag pan. DO NOT call bridge (event filter already
        // sees this press, but Plan 03 Task 1 A6 audit confirmed MMB behavior
        // in Souris mode is aborting — Plan 05 relies on the audit finding
        // that MMB-pan is safe in regular Souris mode OR Ctrl+MMB if not).
        // In either case, setting pan_active_ here and accepting the event
        // prevents QWidget default behavior; the bridge's eventFilter still
        // runs on the WATCHED event BEFORE this method (Qt event order).
        pan_start_            = event->pos();
        pan_anchor_transform_ = state_->view_transform_;
        pan_active_           = true;
        event->accept();
        return;
    }
    QWidget::mousePressEvent(event);
}

void XvueCanvas::mouseMoveEvent(QMouseEvent* event) {
    if (pan_active_ && (event->buttons() & Qt::MiddleButton) && state_) {
        const QPoint delta = event->pos() - pan_start_;
        state_->view_transform_ = pan_anchor_transform_ *
                                  QTransform().translate(delta.x(), delta.y());
        update();
        event->accept();
        return;
    }
    QWidget::mouseMoveEvent(event);
}

void XvueCanvas::mouseReleaseEvent(QMouseEvent* event) {
    if (event->button() == Qt::MiddleButton && pan_active_) {
        pan_active_ = false;
        event->accept();
        return;
    }
    QWidget::mouseReleaseEvent(event);
}

void XvueCanvas::contextMenuEvent(QContextMenuEvent* event) {
    // UI-SPEC §Canvas interaction: right-click opens a module-populated menu.
    // D-08: suppressed when blockingDepth > 0.
    if (XvueApp::blockingDepth() > 0) {
        event->accept();
        return;
    }
    // Count for tests even when menu is empty.
    ++contextMenuShownCount_;

    // Try to reach the menu bridge through the window's accessor. May be
    // nullptr in v1 (populator not registered by module yet) — menu stays empty.
    QMenu menu(this);
    if (auto& win = XvueApp::window_slot()) {
        if (auto* mb = win->menuBridge()) {
            mb->populateContextMenu(&menu);
        }
    }
    if (!menu.isEmpty()) {
        menu.exec(event->globalPos());
    }
    event->accept();
}

void XvueCanvas::resetView() {
    if (!state_) return;
    state_->view_transform_ = QTransform();
    update();
}

**IMPORTANT NOTE on MMB bridge conflict (Research §7 Pitfall 7.2):**
The Qt event order is: `eventFilter(watched=canvas, event)` → `canvas::eventHandler(event)`. So Phase 5's `XvueEventBridge::eventFilter` sees the MiddleButton press BEFORE `XvueCanvas::mousePressEvent` runs.

Phase 5's filter currently (`xvue_qt_event.cpp:329`) treats MiddleButton in Souris mode as abort (sets notypeevent=0, nbc=2, returns true to eat the event). That means if `loop_ != nullptr` (a blocking read IS in progress), the filter eats the middle-click before the canvas's pan handler sees it.

**Consequence:** middle-drag pan works ONLY when NO blocking read is active (`loop_ == nullptr`, which makes the filter return false for ALL events → canvas receives them). This is ACCEPTABLE behavior for v1:
- User is NOT in the mesher picking loop → middle-drag pans (normal desktop UX).
- User IS mid-picking → middle-click aborts (Phase 5 contract preserved).

Plan 03's A6 audit (Task 1) verifies this is X11 parity. If the audit finds that X11 ALSO pans (not aborts) during picking, Plan 03 may file a revision for Plan 05 to use Ctrl+MMB instead. For v1, ship the dual-mode behavior and document in Plan 05 SUMMARY.
