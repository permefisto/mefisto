### testMenuOrder slot (6.2 Plan 04 pattern)

The menu bar must show `{File, Fluid, View, Help}` (not File/Help/Fluid/View
or any other order). The test filters `mb->actions()` on non-null
`QAction::menu()` to avoid counting separator actions:

```cpp
void testMenuOrder() {
    auto* win = XvueApp::window_slot().get();
    QVERIFY(win);
    auto* mb = win->menuBar();
    QVERIFY(mb);
    QList<QMenu*> topMenus;
    for (QAction* a : mb->actions()) {
        if (a->menu()) topMenus.append(a->menu());
    }
    QVERIFY2(topMenus.size() >= 4,
             qPrintable(QStringLiteral("Expected at least 4 top-level menus, got %1")
                        .arg(topMenus.size())));
    QCOMPARE(topMenus[0]->objectName(), QStringLiteral("File"));
    QCOMPARE(topMenus[1]->objectName(), QStringLiteral("Fluid"));
    QCOMPARE(topMenus[2]->objectName(), QStringLiteral("View"));
    QCOMPARE(topMenus[3]->objectName(), QStringLiteral("Help"));
}
```

// xvue/qt/tests/test_xvue_qt_flui_menu.cpp — Phase 6.3 Plan 03
// D-13 flui menu round-trip tests. Offscreen QPA per 6.0 contract.
// Mirrors 6.2's test_xvue_qt_elas_menu.cpp — substitute flui for elas,
// Fluid for Solve; adds testMenuOrder (6.2 Plan 04) and
// testBilingualLabelsEnglish (6.2 Plan 05) as autonomous gates.
#include "test_helpers.h"
#include "xvue_qt_app.h"
#include "xvue_qt_window.h"
#include "xvue_qt_menu_bridge.h"
#include "xvue_qt_menu_file_parser.h"
#include "xvue_qt_i18n.h"

#include <QtTest/QtTest>
#include <QAction>
#include <QCloseEvent>
#include <QFile>
#include <QMenu>
#include <QMenuBar>

extern "C" void xvinitgraphique_(void);
extern "C" void xvfermer_(void);
extern "C" void xvsouris_(int*, int*, int*, int*);
extern "C" void xvue_module_init_(char* name, int* name_len);

class TestXvueQtFluiMenu : public QObject {
    Q_OBJECT

private:
    // Recursive helper — find a QAction whose "lexicon" dynamic property
    // matches `lex`. Walks submenus. Same as 6.1/6.2 pattern.
    static QAction* actionByLexicon(QMenu* menu, const QString& lex) {
        if (!menu) return nullptr;
        for (QAction* a : menu->actions()) {
            if (a->property("lexicon").toString() == lex) return a;
            if (a->menu()) {
                if (auto* child = actionByLexicon(a->menu(), lex)) return child;
            }
        }
        return nullptr;
    }

    // Drain chars via xvsouris_ and assert the sequence.
    static void expectChars(const char* expected, int n) {
        for (int i = 0; i < n; ++i) {
            int nt = 0, nbc = 0, x = 0, y = 0;
            xvsouris_(&nt, &nbc, &x, &y);
            QCOMPARE(nt, 2);
            QCOMPARE(nbc, static_cast<int>(static_cast<unsigned char>(expected[i])));
        }
    }

private slots:
    void initTestCase() {
        qunsetenv("MEFISTO_XVSOURIS_AUTOEXIT");
        xvinitgraphique_();
        char name[] = "flui";
        int  nlen   = 4;
        xvue_module_init_(name, &nlen);  // registerFluiActions_stub_ runs
    }

    void cleanupTestCase() {
        auto* win = XvueApp::window_slot().get();
        if (win && win->menuBridge()) {
            while (win->menuBridge()->popChar().has_value()) {}
        }
        xvfermer_();
    }

    // D-13 test 1: Fluid menu -> Steady Stokes (5;)
    void testFluidSolveStatic() {
        auto* win = XvueApp::window_slot().get();
        QVERIFY(win && win->menuBridge());
        auto* fluidMenu = win->menuBar()->findChild<QMenu*>(QStringLiteral("Fluid"));
        QVERIFY(fluidMenu);
        auto* act = actionByLexicon(fluidMenu, QStringLiteral("5;"));
        QVERIFY(act);
        act->trigger();
        const char exp[3] = { '5', ';', 13 };
        expectChars(exp, 3);
    }

    // D-13 test 2: View menu -> Draw velocities/pressures (10;)
    void testViewDrawResults() {
        auto* win = XvueApp::window_slot().get();
        auto* viewMenu = win->menuBar()->findChild<QMenu*>(QStringLiteral("View"));
        QVERIFY(viewMenu);
        auto* act = actionByLexicon(viewMenu, QStringLiteral("10;"));
        QVERIFY(act);
        act->trigger();
        const char exp[4] = { '1', '0', ';', 13 };
        expectChars(exp, 4);
    }

    // D-13 test 3: File menu -> MS TMS File Tools (70;)
    void testFileMSTools() {
        auto* win = XvueApp::window_slot().get();
        auto* fileMenu = win->menuBar()->findChild<QMenu*>(QStringLiteral("File"));
        QVERIFY(fileMenu);
        auto* act = actionByLexicon(fileMenu, QStringLiteral("70;"));
        QVERIFY(act);
        act->trigger();
        const char exp[4] = { '7', '0', ';', 13 };
        expectChars(exp, 4);
    }

    // D-13 test 4: Help menu -> no lexicon payload (shared 6.0 actions)
    void testHelpNoQueue() {
        auto* win = XvueApp::window_slot().get();
        auto* help = win->menuBar()->findChild<QMenu*>(QStringLiteral("Help"));
        QVERIFY(help);
        const int before = win->menuBridge()->queueSize();
        for (QAction* a : help->actions()) {
            QVERIFY2(a->property("lexicon").toString().isEmpty(),
                     qPrintable(QStringLiteral("help action %1 has lexicon payload %2")
                                .arg(a->text(), a->property("lexicon").toString())));
        }
        QCOMPARE(win->menuBridge()->queueSize(), before);
    }

    // Bonus: D-09 closeEvent pushes 99; when blocking (mirrors 6.2 Case 5).
    void testCloseEventPushes99WhenBlocking() {
        auto* win = XvueApp::window_slot().get();
        QVERIFY(win && win->menuBridge());
        while (win->menuBridge()->popChar().has_value()) {}
        const int before = win->menuBridge()->queueSize();
        {
            BlockingDepthGuard g;  // depth > 0 in this scope
            QCloseEvent ev;
            win->closeEvent(&ev);
            QVERIFY(!ev.isAccepted());
        }
        const int after = win->menuBridge()->queueSize();
        QCOMPARE(after - before, 4);  // '9','9',';',CR
    }

    // testMenuOrder: {File, Fluid, View, Help} — codifies Plan 04 fix
    // from 6.2 as an automated gate (replaces the 6.2 manual A/B checkpoint).
    void testMenuOrder() {
        auto* win = XvueApp::window_slot().get();
        QVERIFY(win);
        auto* mb = win->menuBar();
        QVERIFY(mb);
        QList<QMenu*> topMenus;
        for (QAction* a : mb->actions()) {
            if (a->menu()) topMenus.append(a->menu());
        }
        QVERIFY2(topMenus.size() >= 4,
                 qPrintable(QStringLiteral("Expected at least 4 top menus, got %1")
                            .arg(topMenus.size())));
        QCOMPARE(topMenus[0]->objectName(), QStringLiteral("File"));
        QCOMPARE(topMenus[1]->objectName(), QStringLiteral("Fluid"));
        QCOMPARE(topMenus[2]->objectName(), QStringLiteral("View"));
        QCOMPARE(topMenus[3]->objectName(), QStringLiteral("Help"));
    }

    // testBilingualLabelsEnglish: EN labels from td/ma/debuflui when
    // anglais flag present — codifies Plan 05 fix as an automated gate.
    void testBilingualLabelsEnglish() {
        const QString mefisto = qEnvironmentVariable("MEFISTO");
        if (mefisto.isEmpty()) QSKIP("MEFISTO env not set");
        if (!QFile::exists(mefisto + QStringLiteral("/td/m/anglais")))
            QSKIP("anglais flag not set (FR install)");
        if (!QFile::exists(mefisto + QStringLiteral("/td/ma/debuflui")))
            QSKIP("td/ma/debuflui not found");
        XvueMenuFileParser::clearCacheForTesting();
        const MenuFile& mf = XvueMenuFileParser::loadFor(QStringLiteral("debuflui"));
        QVERIFY(mf.ok());
        const QString label5 = mf.label(5);
        QVERIFY2(label5.contains(QStringLiteral("STOKES"), Qt::CaseInsensitive)
                 || label5.contains(QStringLiteral("INCOMPRESSIBLE"), Qt::CaseInsensitive),
                 qPrintable(QStringLiteral("EN label for 5;: ") + label5));
        const QString label99 = mf.label(99);
        QVERIFY2(label99.contains(QStringLiteral("SAVE"), Qt::CaseInsensitive)
                 || label99.contains(QStringLiteral("QUIT"), Qt::CaseInsensitive),
                 qPrintable(QStringLiteral("EN label for 99;: ") + label99));
        XvueMenuFileParser::clearCacheForTesting();
    }
};

int main(int argc, char* argv[]) {
    (void)argc; (void)argv;
    // 6.1 auto-fix #3: disable stdout redirect so QtTest totals reach terminal.
    qputenv("XVUE_QT_DISABLE_STDOUT_REDIRECT", "1");
    XvueApp::ensure();
    TestXvueQtFluiMenu tc;
    int qt_argc = 1;
    char qt_arg0[] = "xvue_qt_flui_menu_tests";
    char* qt_argv[] = { qt_arg0, nullptr };
    return QTest::qExec(&tc, qt_argc, qt_argv);
}
