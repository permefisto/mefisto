          ```cpp
          // Helper for top-level menu creation with a stable objectName.
          // When `insertBefore` is non-null, the new menu is inserted
          // BEFORE `insertBefore` via QMenuBar::insertMenu(action, menu).
          // When null (or insertBefore is not in the menuBar), falls back
          // to addMenu (append).
          QMenu* ensureTopLevelMenu(QMenuBar* mbar,
                                    const QString& title,
                                    const QString& objectName,
                                    QMenu* insertBefore = nullptr)
          {
              if (auto* existing = mbar->findChild<QMenu*>(objectName)) {
                  return existing;
              }
              auto* m = new QMenu(title, mbar);
              m->setObjectName(objectName);
              if (insertBefore) {
                  mbar->insertMenu(insertBefore->menuAction(), m);
              } else {
                  mbar->addMenu(m);
              }
              return m;
          }
          ```
          NOTE: We construct `QMenu(title, mbar)` directly (not via `mbar->addMenu(title)`) because `addMenu(QString)` both creates AND appends in one call — there's no way to create-without-appending. Manually constructing gives us control over insertion site. The QMenu parent is `mbar` so Qt's parent-child tree handles deletion.
