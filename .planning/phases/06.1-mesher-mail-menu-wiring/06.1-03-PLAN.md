```cpp
void XvueWindow::closeEvent(QCloseEvent* event) {
    // Persist geometry/state as 6.0 does today.
    XvuePrefs::saveWindowGeometry(saveGeometry());
    XvuePrefs::saveWindowState(saveState());
    if (consoleDock_) {

    // D-09 + RESEARCH Pitfall 3: only intercept when Fortran is
```

```cpp
#include <QMenuBar>

extern "C" void xvinitgraphique_(void);
extern "C" void xvfermer_(void);
extern "C" void xvsouris_(int*, int*, int*, int*);
extern "C" void xvue_module_init_(char* name, int* name_len);

class TestXvueQtMailMenu : public QObject {
    Q_OBJECT

private:
    // Recursive helper — find a QAction whose "lexicon" dynamic
    // property matches `lex`. Matches Plan 02's makeLeafAction
    // property naming. Walks submenus.
    static QAction* actionByLexicon(QMenu* menu, const QString& lex) {
        if (!menu) return nullptr;
        for (QAction* a : menu->actions()) {
            if (a->property("lexicon").toString() == lex) return a;
            if (a->menu()) {
                if (auto* child = actionByLexicon(a->menu(), lex)) return child;
            }
        }
        return nullptr;
    }

    // Drain chars via xvsouris_ and assert the sequence. Each read is
    // a WaitMode::Souris call returning notypeevent=2, nbc=<ascii>.
    static void expectChars(const char* expected, int n) {
        for (int i = 0; i < n; ++i) {
            int nt = 0, nbc = 0, x = 0, y = 0;
            xvsouris_(&nt, &nbc, &x, &y);
            QCOMPARE(nt, 2);
            QCOMPARE(nbc, static_cast<int>(static_cast<unsigned char>(expected[i])));
        }
    }

private slots:
    void initTestCase() {
        // 6.0 + 6.1 require offscreen QPA. The test runner env sets it.
        qunsetenv("MEFISTO_XVSOURIS_AUTOEXIT");
        xvinitgraphique_();
        char name[] = "mail";
        int  nlen   = 4;
        xvue_module_init_(name, &nlen);  // registerMailActions_stub_ runs
    }

    void cleanupTestCase() {
        auto* win = XvueApp::window_slot().get();
        if (win && win->menuBridge()) {
            while (win->menuBridge()->popChar().has_value()) {}
        }
        xvfermer_();
    }

    // D-13 test 1: Mesh → Points → Create  (1;1;)
    void testMeshPointsCreate() {
        auto* win = XvueApp::window_slot().get();
        QVERIFY(win && win->menuBridge());
        auto* meshMenu = win->menuBar()->findChild<QMenu*>(QStringLiteral("Mesh"));
        QVERIFY(meshMenu);
        auto* act = actionByLexicon(meshMenu, QStringLiteral("1;1;"));
        QVERIFY(act);
        act->trigger();
        const char exp[5] = { '1', ';', '1', ';', 13 };
        expectChars(exp, 5);
    }

    // D-13 test 2: File → Import PLSVO  (80;)
    void testFileImport() {
        auto* win = XvueApp::window_slot().get();
        auto* fileMenu = win->menuBar()->findChild<QMenu*>(QStringLiteral("File"));
        QVERIFY(fileMenu);
        auto* act = actionByLexicon(fileMenu, QStringLiteral("80;"));
        QVERIFY(act);
        act->trigger();
        const char exp[4] = { '8', '0', ';', 13 };
        expectChars(exp, 4);
    }

    // D-13 test 3: View → Draw PLSVO meshes  (10;)
    void testViewDrawMesh() {
        auto* win = XvueApp::window_slot().get();
        auto* viewMenu = win->menuBar()->findChild<QMenu*>(QStringLiteral("View"));
        QVERIFY(viewMenu);
        auto* act = actionByLexicon(viewMenu, QStringLiteral("10;"));
        QVERIFY(act);
        act->trigger();
        const char exp[4] = { '1', '0', ';', 13 };
        expectChars(exp, 4);
    }

    // D-13 test 4: Help → no lexicon payload (shared 6.0 action)
    void testHelpAboutNoQueue() {
        auto* win = XvueApp::window_slot().get();
        auto* help = win->menuBar()->findChild<QMenu*>(QStringLiteral("Help"));
        QVERIFY(help);
        const int before = win->menuBridge()->queueSize();
        for (QAction* a : help->actions()) {
            QVERIFY2(a->property("lexicon").toString().isEmpty(),
                     qPrintable(QStringLiteral("help action %1 has lexicon payload %2")
                                .arg(a->text(), a->property("lexicon").toString())));
        }
        QCOMPARE(win->menuBridge()->queueSize(), before);
    }

    // Bonus: D-09 closeEvent pushes 99; when blocking. Uses a scripted
    // blockingDepth mutation if available; otherwise skip gracefully.
    void testCloseEventPushes99WhenBlocking() {
        auto* win = XvueApp::window_slot().get();
        QVERIFY(win && win->menuBridge());
        // Drain any existing chars.
        while (win->menuBridge()->popChar().has_value()) {}
        const int before = win->menuBridge()->queueSize();

        // If XvueApp exposes a testing-only depth bump, use it. Else

        // validated 99; typed at prompt exits cleanly per 6.0 Plan 07).
        // The existing RAII guard in xvue_qt_event.cpp is the production
        // depth incrementer; we emulate it via the accessor.
#ifdef XVUE_QT_TESTING_BLOCKING_DEPTH
        XvueApp::pushBlockingDepthForTesting();
#else
        // Without a testing hook, depth is 0 here — closeEvent would
        // fall through. Skip this test until the hook lands.
        QSKIP("blockingDepth testing hook not available; covered by manual A/B gate");
#endif
        QCloseEvent ev;
        win->closeEvent(&ev);
        QVERIFY(!ev.isAccepted());
        const int after = win->menuBridge()->queueSize();
        QCOMPARE(after - before, 4);  // '9','9',';',CR

        XvueApp::popBlockingDepthForTesting();
#endif
```

```cpp
void XvueWindow::onFileQuit() {
    // D-09 silent dispatch: push "99;" into the menu bridge; Fortran's
    // main LIMTCL loop drains it, DONNMF dispatches NMTCL=99, and
    // ppmail.f label 9900 calls ARRET(0) → STOP → clean process exit.
    // No QMessageBox confirm — matches typed-lexicon "99;" which is
    // also unconfirmed.
    if (menuBridge_) {
        if (consoleDock_) {
            consoleDock_->appendLine(QStringLiteral("[menu] 99;"));
        }
        menuBridge_->queueLexicon(QStringLiteral("99;"));
    }
    // Deliberately NO close() / QCoreApplication::quit() — Fortran
    // owns the exit.
}
```

Replace the existing `closeEvent` body with the blocking-depth-guarded
version (keep the geometry/state persist logic that 6.0 put there):

```cpp
void XvueWindow::closeEvent(QCloseEvent* event) {
    // Persist window state — 6.0 contract preserved.
    XvuePrefs::saveWindowGeometry(saveGeometry());
    XvuePrefs::saveWindowState(saveState());
    if (consoleDock_) {
        XvuePrefs::saveConsoleVisible(consoleDock_->isVisible());
    }

    // D-09 + RESEARCH Pitfall 3: only intercept while Fortran is in
    // a blocking xvsouris_ read. If depth == 0, let the default close
    // proceed — Fortran has already exited, or we're in batch mode,
    // or we're in the atexit teardown from QApplication destruction.
    if (menuBridge_ && XvueApp::blockingDepth() > 0) {
        if (consoleDock_) {
            consoleDock_->appendLine(QStringLiteral("[menu] 99;"));
        }
        menuBridge_->queueLexicon(QStringLiteral("99;"));
        event->ignore();
        return;
    }

    QMainWindow::closeEvent(event);
}
```
