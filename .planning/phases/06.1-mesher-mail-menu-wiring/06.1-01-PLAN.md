```python
#!/usr/bin/env python3
"""
tools/validate_audit_md.py — schema validator for LEXICON-AUDIT-*.md.

Enforces the 9 rules in .planning/phases/06.1-*/06.1-RESEARCH.md
§Validation Architecture > Schema validator.

Usage:
    python3 tools/validate_audit_md.py <path-to-audit.md>

Exit codes:
    0 = all rules pass
    1 = schema violation (first violation printed to stderr)
"""
import re
import sys
from pathlib import Path

EXPECTED_COLS = [
    "lexicon_path", "description_fr", "description_en", "frequency",
    "qaction", "toolbar", "icon_source", "shortcut", "notes",
]
FREQ_OK = {"high", "med", "low", "—"}
BOOL_OK = {"yes", "no"}
LEX_RX = re.compile(r"^\d+;(?:\d+;)*$")
ROW_BOUND = (80, 250)
TOOLBAR_YES_EXACT = 5
ICONS_DIR = Path(__file__).resolve().parent.parent / "xvue" / "qt" / "resources" / "icons" / "mail"


def fail(msg):
    sys.stderr.write(f"validate_audit_md: {msg}\n")
    sys.exit(1)


def rows(path):
    in_table = False
    header = None
    for line in path.read_text(encoding="utf-8").splitlines():
        if line.startswith("|") and not in_table:
            header = [c.strip() for c in line.strip("|").split("|")]
            in_table = True
            continue
        if in_table:
            if not line.startswith("|"):
                break
            if set(line.replace("|", "").strip()) <= set("-: "):
                continue
            yield header, [c.strip() for c in line.strip("|").split("|")]


def main():
    if len(sys.argv) != 2:
        fail("usage: validate_audit_md.py <audit.md>")
    p = Path(sys.argv[1])
    if not p.is_file():
        fail(f"not a file: {p}")

    header = None
    data = []
    for h, r in rows(p):
        header = h
        data.append(r)

    # Rule 1 — 9 columns, exact schema
    if header != EXPECTED_COLS:
        fail(f"rule 1 (9-col schema) failed: got {header}")

    # Rule 2 — row count bound
    lo, hi = ROW_BOUND
    if not (lo <= len(data) <= hi):
        fail(f"rule 2 (row count in [{lo},{hi}]) failed: got {len(data)}")

    # Rules 3/4/5/6 — per-row enum and regex checks
    for i, row in enumerate(data, start=1):
        if len(row) != 9:
            fail(f"rule 0 (9 cols per row) failed at row {i}: {row}")
        lex, df, en, freq, qa, tb, icon, sc, notes = row
        if not LEX_RX.match(lex):
            fail(f"rule 3 (lexicon_path regex) failed at row {i}: {lex!r}")
        if freq not in FREQ_OK:
            fail(f"rule 4 (frequency enum) failed at row {i}: {freq!r}")
        if qa not in BOOL_OK:
            fail(f"rule 5 (qaction enum) failed at row {i}: {qa!r}")
        if tb not in BOOL_OK:
            fail(f"rule 6 (toolbar enum) failed at row {i}: {tb!r}")

    # Rule 7 — exactly 5 toolbar=yes
    n_tb = sum(1 for r in data if r[5] == "yes")
    if n_tb != TOOLBAR_YES_EXACT:
        fail(f"rule 7 (toolbar=yes == 5) failed: got {n_tb}")

    # Rule 8 — qaction=yes implies frequency in {high, med}
    for i, r in enumerate(data, start=1):
        if r[4] == "yes" and r[3] not in {"high", "med"}:
            fail(f"rule 8 (qaction=yes => freq in high/med) failed at row {i}: freq={r[3]!r}")

    # Rule 9 — icon_source cells ending .svg have matching file under
    # xvue/qt/resources/icons/mail/ (cross-check with Plan 02 deliverable).
    # Plan 01 may run before Plan 02 commits its icons; allow missing
    # files as WARN (not FAIL) by default. Toggle via env var.
    import os
    strict_icons = os.environ.get("AUDIT_STRICT_ICONS", "0") == "1"
    for i, r in enumerate(data, start=1):
        icon = r[6]
        if icon.endswith(".svg"):
            f = ICONS_DIR / icon
            if not f.exists():
                msg = f"rule 9 (svg exists) failed at row {i}: {f}"
                if strict_icons:
                    fail(msg)
                else:
                    sys.stderr.write(f"WARN: {msg}\n")

    print(f"OK — {len(data)} rows, {n_tb} toolbar=yes, schema valid")
    sys.exit(0)


if __name__ == "__main__":
    main()
```
