**Entry 5 — `xvfond_` (SHELL-05, D-14, D-15)**

```cpp
void proc(xvfond)(int *icolor) {
    XvueApp::ensure();
    XVUE_QT_ASSERT_MAIN_THREAD();
    if (!icolor) return;

    // D-14: Phase 1 has no palette yet. Minimal 2-entry mapping matches the
    // legacy X11 BlackPixel/WhitePixel convention (xvuelc.c:935 et seq.).
    QColor chosen = Qt::black;
    if (*icolor == 0) {
        chosen = Qt::black;
    } else if (*icolor == 1) {
        chosen = Qt::white;
    } else {
        static bool warned_xvfond_range = false;
        if (!warned_xvfond_range) {
            std::fprintf(stderr,
                "xvue-qt: xvfond_ palette index %d out of Phase 1 range "
                "(Phase 3 will add full colormap)\n", *icolor);
            warned_xvfond_range = true;
        }
        chosen = Qt::black;
    }

    // D-15: update XvueState::background_ through the live window, schedule
    // repaint. If no window is open the color is still latched on the next
    // xvinitgraphique_ via a fresh XvueState — but since Phase 1 XvueState is
    // owned by XvueWindow, there is nowhere to store it until a window exists.
    // Document this explicitly: with no open window, xvfond_ is a no-op past
    // the warn-once line. Phase 2+ may revisit if needed.
    auto& win = XvueApp::window_slot();
    if (win) {
        win->state()->background_ = chosen;
        if (win->canvas()) {
            win->canvas()->update();
        }
    }
}
```
